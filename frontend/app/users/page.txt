        'use client';

        import { useEffect, useState } from 'react';
        import { Button, Card, Table } from '@gravity-ui/uikit';
        import { useRouter } from 'next/navigation';

        const API =
        process.env.NEXT_PUBLIC_API_URL || 'http://localhost:3001';

        type User = {
        id: string;
        name: string;
        email: string;
        role: 'SUPER_ADMIN' | 'ADMIN' | 'USER';
        tenant?: {
            id: string;
            name: string;
            slug: string;
        } | null;
        createdAt?: string;
        };

        export default function UsersPage() {
        const router = useRouter();

        const [users, setUsers] = useState<User[]>([]);
        const [loading, setLoading] = useState(true);
        const [error, setError] = useState('');

        async function loadUsers() {
            try {
            setLoading(true);
            setError('');

            const res = await fetch(`${API}/users`, {
                method: 'GET',
                credentials: 'include',
            });

            if (res.status === 401) {
                router.replace('/');
                return;
            }

            if (res.status === 403) {
                setError('Anda tidak memiliki akses ke User Management.');
                return;
            }

            if (!res.ok) {
                throw new Error('Gagal mengambil data user');
            }

            const data = await res.json();

            setUsers(Array.isArray(data) ? data : data.users || []);
            } catch (err) {
            setError(
                err instanceof Error
                ? err.message
                : 'Terjadi kesalahan'
            );
            } finally {
            setLoading(false);
            }
        }

        useEffect(() => {
            loadUsers();
        }, []);

        return (
            <main className="cti-content">

            <div className="cti-title-row">
                <div>
                <h1>User Management</h1>
                <p>
                    Kelola pengguna dan role dalam platform CTI
                </p>
                </div>

                <Button
                view="action"
                size="l"
                >
                + Add User
                </Button>
            </div>

            {error && (
                <Card
                view="outlined"
                className="cti-activity"
                >
                <p>{error}</p>
                </Card>
            )}

            <Card
                view="outlined"
                className="cti-activity"
            >

                {loading ? (
                <p>Loading users...</p>
                ) : users.length === 0 ? (
                <p>Belum ada user.</p>
                ) : (
                <Table
                    data={users}
                    columns={[
                    {
                        id: 'name',
                        name: 'Name',
                        template: (user: User) => user.name,
                    },
                    {
                        id: 'email',
                        name: 'Email',
                        template: (user: User) => user.email,
                    },
                    {
                        id: 'role',
                        name: 'Role',
                        template: (user: User) => user.role,
                    },
                    {
                        id: 'tenant',
                        name: 'Tenant',
                        template: (user: User) =>
                        user.tenant?.name || 'Global',
                    },
                    {
                        id: 'actions',
                        name: 'Actions',
                        template: () => (
                        <div
                            style={{
                            display: 'flex',
                            gap: 8,
                            }}
                        >
                            <Button
                            view="flat"
                            size="s"
                            >
                            Edit
                            </Button>

                            <Button
                            view="flat-danger"
                            size="s"
                            >
                            Deactivate
                            </Button>
                        </div>
                        ),
                    },
                    ]}
                />
                )}

            </Card>

            </main>
        );
        }
